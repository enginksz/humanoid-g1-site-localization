# ROS-free build of hku-mars/FAST_LIO (FAST-LIO2) for the simulator sequences.
cmake_minimum_required(VERSION 3.16)
project(fast_lio_standalone C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# same flags as upstream minus -march=native (Eigen/PCL ABI, see fmcw_lio_standalone)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
add_definitions(-w -DMP_EN -DMP_PROC_NUM=4)
set(UP ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/FAST_LIO)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(OpenMP REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(fast_lio_offline src/laserMapping_offline.cpp ${UP}/include/ikd-Tree/ikd_Tree.cpp)
target_compile_definitions(fast_lio_offline PRIVATE ROOT_DIR="${UP}/")
target_include_directories(fast_lio_offline PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/shim ${UP}/include ${UP}/src
  ${PCL_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIR})
target_link_libraries(fast_lio_offline PRIVATE ${PCL_LIBRARIES} OpenMP::OpenMP_CXX yaml-cpp pthread)
